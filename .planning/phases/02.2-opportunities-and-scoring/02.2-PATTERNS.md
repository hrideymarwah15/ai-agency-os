```python
@dataclass
class Rule:
    name: str
    category: str  # "automation" | "ai"
    condition: Callable[[Business, WebsiteAudit], bool]
    recommendation: str
    roi_params: dict[str, Any]  # time_saved_hours_monthly, frequency_monthly, impl_cost_estimate
    confidence: float

class RuleEngine:
    def __init__(self, rules_path: str):
        self.rules_path = rules_path
        self._rules: list[Rule] = []
        self._mtime = 0
        self.load_rules()

    def load_rules(self) -> None:
        """Parse YAML, compile condition lambdas, watch mtime."""
        config = load_rules(self.rules_path)
        self._rules = self._compile_rules(config)
        self._mtime = os.path.getmtime(self.rules_path)

    def _compile_rules(self, config: RulesConfig) -> list[Rule]:
        """Convert RuleModel → Rule with callable conditions."""
        rules = []
        for rm in config.automation_rules + config.ai_rules:
            condition = self._build_condition(rm)
            rules.append(Rule(
                name=rm.name,
                category=rm.category,
                condition=condition,
                recommendation=rm.recommendation,
                roi_params=rm.roi_params,
                confidence=rm.confidence,
            ))
        return rules

    def _build_condition(self, rm: RuleModel) -> Callable[[Business, WebsiteAudit], bool]:
        """Compile detection dict → predicate (business, audit) -> bool."""
        det = rm.detection or {}
        industry_list = [i.lower() for i in (rm.industry or [])]

        def check(business: Business, audit: WebsiteAudit) -> bool:
            if audit is None or audit.status != "COMPLETED":
                return False
            # Industry match (automation rules)
            if industry_list and (business.industry or "").lower() not in industry_list:
                return False
            # tech_stack_excludes
            excludes = det.get("tech_stack_excludes", [])
            if excludes:
                audit_tech = {t["name"].lower() for t in (audit.tech_stack or [])}
                if any(ex.lower() in audit_tech for ex in excludes):
                    return False
            # gaps_contains
            gaps_needed = det.get("gaps_contains", [])
            if gaps_needed:
                audit_gaps = {g["kind"] for g in (audit.gaps or [])}
                if not all(g in audit_gaps for g in gaps_needed):
                    return False
            # content_check (ai rules)
            if det.get("content_check") == "faq_keyword_absent":
                # placeholder — static analyzer would set a signal
                pass
            return True
        return check

    def evaluate_automation(self, business: Business, audit: WebsiteAudit) -> list[dict]:
        """Return list of automation opportunity dicts with ROI, confidence."""
        if audit is None or audit.status != "COMPLETED":
            return []
        results = []
        for rule in self._rules:
            if rule.category == "automation" and rule.condition(business, audit):
                results.append({
                    "name": rule.name,
                    "category": "automation",
                    "recommendation": rule.recommendation,
                    "roi_params": rule.roi_params,
                    "confidence": rule.confidence,
                    "source_signals": {"pattern": rule.name},
                })
        return results

    def evaluate_ai(self, business: Business, audit: WebsiteAudit) -> list[dict]:
        """Return list of AI opportunity dicts with ROI, confidence."""
        if audit is None or audit.status != "COMPLETED":
            return []
        results = []
        for rule in self._rules:
            if rule.category == "ai" and rule.condition(business, audit):
                results.append({
                    "name": rule.name,
                    "category": "ai",
                    "recommendation": rule.recommendation,
                    "roi_params": rule.roi_params,
                    "confidence": rule.confidence,
                    "source_signals": {"signal": rule.name},
                })
        return results
```

```python
class ROICalculator:
    def __init__(self, hourly_buckets: dict[str, dict[str, float]], impl_templates: dict[str, float]):
        self.hourly_buckets = hourly_buckets
        self.impl_templates = impl_templates

    def calc_roi(self, opp: dict, business: Business, audit: WebsiteAudit) -> float:
        """Return monthly ROI in INR (can be negative)."""
        time_saved = opp["roi_params"]["time_saved_hours_monthly"]
        freq = opp["roi_params"]["frequency_monthly"]
        impl_cost = opp["roi_params"].get("impl_cost_estimate") or \
                    self.impl_templates.get(opp["name"], 0)
        industry = (business.industry or "Unknown").strip()
        size_bucket = self._size_bucket(business.employee_count)
        hourly_cost = self.hourly_buckets.get(industry, {}).get(size_bucket, 1000)
        return (time_saved * hourly_cost * freq) - impl_cost

    def _size_bucket(self, employee_count: int | None) -> str:
        if not employee_count:
            return "Micro"
        if employee_count <= 10: return "Micro"
        if employee_count <= 50: return "Small"
        if employee_count <= 200: return "Medium"
        if employee_count <= 1000: return "Large"
        return "Enterprise"
```

```python
@dataclass
class ScoreBreakdown:
    factor: str
    weight_pct: int
    raw_score: int
    weighted_score: float
    signals_used: list[str]

class LeadScorer:
    def __init__(self, weights: dict[str, float]):
        # weights sum to 1.0; keys per D-52
        self.weights = weights

    def score_lead(
        self,
        business: Business,
        audit: WebsiteAudit | None,
        opportunities: list[dict]
    ) -> tuple[int, str, list[ScoreBreakdown]]:
        """Return (score_0_100, tier, breakdown_list)."""
        if audit is None or audit.status != "COMPLETED":
            return 0, "Raw", [
                ScoreBreakdown(f, int(w*100), 0, 0.0, ["no_audit"])
                for f, w in self.weights.items()
            ]

        breakdown = []
        total = 0.0
        for factor, weight in self.weights.items():
            raw = self._score_factor(factor, business, audit, opportunities)
            weighted = raw * weight
            total += weighted
            breakdown.append(ScoreBreakdown(
                factor, int(weight*100), raw, weighted, []
            ))

        score = max(0, min(100, round(total)))
        tier = self._tier(score)
        return score, tier, breakdown

    def _tier(self, score: int) -> str:
        if score >= 80: return "Hot"
        if score >= 60: return "Warm"
        if score >= 40: return "Cold"
        return "Raw"

    # ---- Factor implementations (RESEARCH Code Examples lines 741-787) ----
    def _score_factor(self, factor: str, business: Business, audit: WebsiteAudit, opps: list) -> int:
        dispatch = {
            "website_quality": self._score_website_quality,
            "seo_maturity": self._score_seo_maturity,
            "growth_signals": self._score_growth_signals,
            "company_size": self._score_company_size,
            "automation_potential": self._score_automation_potential,
            "ai_potential": self._score_ai_potential,
            "budget_estimate": self._score_budget_estimate,
        }
        return dispatch[factor](business, audit, opps)

    def _score_website_quality(self, business: Business, audit: WebsiteAudit, opps: list) -> int:
        scores = [audit.performance_score, audit.accessibility_score, audit.seo_score, audit.mobile_score]
        available = [s for s in scores if s is not None]
        return round(sum(available) / len(available)) if available else 0

    def _score_seo_maturity(self, business: Business, audit: WebsiteAudit, opps: list) -> int:
        base = audit.seo_score or 0
        seo_gaps = sum(1 for g in (audit.gaps or []) if g.get("kind") in ("missing_cta", "no_direct_contact", "missing_contact_form"))
        has_analytics = any(t["name"] in ("ga4", "gtm", "ga") for t in (audit.analytics or []))
        return max(0, min(100, base - 10 * seo_gaps + (5 if has_analytics else 0)))

    def _score_growth_signals(self, business: Business, audit: WebsiteAudit, opps: list) -> int:
        score = 30
        for sig in (business.growth_signals or []):
            if sig in ("hiring", "funding", "traffic_growth"): score += 15
            elif sig in ("social_growth", "new_location"): score += 10
        if business.rating and business.rating >= 4.0: score += 10
        if business.review_count and business.review_count > 50: score += 5
        return min(100, score)

    def _score_company_size(self, business: Business, audit: WebsiteAudit, opps: list) -> int:
        emp = business.employee_count or 0
        if 10 <= emp <= 200: return 100
        if emp < 10: return 60
        if emp <= 500: return 80
        return 70

    def _score_automation_potential(self, business: Business, audit: WebsiteAudit, opps: list) -> int:
        auto_opps = [o for o in opps if o["category"] == "automation"]
        manual_gaps = sum(1 for g in (audit.gaps or []) if g.get("kind") in ("missing_contact_form", "form_no_action"))
        return min(100, 30 + 15 * len(auto_opps) + 10 * manual_gaps)

    def _score_ai_potential(self, business: Business, audit: WebsiteAudit, opps: list) -> int:
        ai_opps = [o for o in opps if o["category"] == "ai"]
        has_chatbot = any(t["name"] in ("intercom", "drift", "crisp", "tawk") for t in (audit.analytics or []))
        has_crm = any(t["name"] in ("hubspot", "salesforce", "pipedrive") for t in (audit.analytics or []))
        score = 20 + 20 * len(ai_opps)
        if not has_chatbot: score += 15
        if not has_crm: score += 10
        return min(100, score)

    def _score_budget_estimate(self, business: Business, audit: WebsiteAudit, opps: list) -> int:
        budget = business.estimated_budget or 0
        if budget >= 50000: return 100
        if budget >= 20000: return 70
        if budget >= 10000: return 50
        return 30
```

```python
@dataclass
class LeadView:
    business_id: UUID
    name: str
    industry: str | None
    priority_score: int
    tier: str
    total_roi: float
    top_opportunities: list[dict]  # top 3 by ROI
    last_audited_at: datetime | None
    pipeline_stage: str  # from Business.status

class QueueBuilder:
    def __init__(self, session: AsyncSession):
        self.session = session

    async def build_queue(
        self,
        *,
        tier_filter: str | None = None,
        industry_filter: str | None = None,
        min_score: int | None = None,
        limit: int = 50
    ) -> list[LeadView]:
        """Ordered pipeline view with optional filters."""
        # Subquery: latest completed audit per business
        latest_audit = (
            select(WebsiteAudit.business_id, func.max(WebsiteAudit.started_at).label("latest"))
            .where(WebsiteAudit.status == "COMPLETED")
            .group_by(WebsiteAudit.business_id)
            .subquery()
        )

        stmt = (
            select(Business, latest_audit.c.latest)
            .outerjoin(latest_audit, Business.id == latest_audit.c.business_id)
            .order_by(
                desc(Business.priority_score),
                desc(Business.total_roi),
                desc(Business.updated_at)
            )
            .limit(limit)
        )

        if tier_filter:
            stmt = stmt.where(Business.tier == tier_filter)
        if industry_filter:
            stmt = stmt.where(Business.industry == industry_filter)
        if min_score is not None:
            stmt = stmt.where(Business.priority_score >= min_score)

        rows = (await self.session.execute(stmt)).all()
        return [self._to_view(business, latest) for business, latest in rows]

    def _to_view(self, business: Business, last_audited_at: datetime | None) -> LeadView:
        opps = business.opportunities or []
        top3 = opps[:3]
        return LeadView(
            business_id=business.id,
            name=business.name,
            industry=business.industry,
            priority_score=int(business.priority_score or 0),
            tier=business.tier or "Raw",
            total_roi=business.total_roi or 0.0,
            top_opportunities=[
                {"name": o["recommendation"], "roi_display": f"₹{o['roi_monthly_inr']:,.0f}/mo", "confidence": o["confidence"]}
                for o in top3
            ],
            last_audited_at=last_audited_at,
            pipeline_stage=business.status.value,
        )
```

```python
_OPPORTUNITY_TOPIC = "audit.completed"
_POLL_INTERVAL_SECONDS = 5.0
_MAX_ERROR_LENGTH = 1000

async def opportunity_worker(*, poll_interval: float = _POLL_INTERVAL_SECONDS) -> None:
    """Drain audit.completed events → rescore lead → upsert opportunities/score."""
    while True:
        try:
            await _process_one()
        except asyncio.CancelledError:
            raise
        except Exception:  # noqa: BLE001
            logger.exception("opportunity worker iteration failed")
        await asyncio.sleep(poll_interval)

async def _process_one() -> bool:
    session_maker = database.async_session_maker
    if session_maker is None:
        raise RuntimeError("Database not initialized")
    async with session_maker() as session:
        result = await session.execute(
            select(OutboxEvent)
            .where(OutboxEvent.topic == _OPPORTUNITY_TOPIC, OutboxEvent.status == "PENDING")
            .order_by(OutboxEvent.created_at)
            .limit(1)
        )
        event = result.scalars().first()
        if event is None:
            return False
        event.status = "IN_PROGRESS"
        event.attempts += 1
        await session.commit()
        try:
            business_id = UUID(event.payload["business_id"])
            await rescore_lead(session, business_id)
        except Exception as exc:  # noqa: BLE001
            event.status = "FAILED"
            event.last_error = str(exc)[:_MAX_ERROR_LENGTH]
            event.processed_at = utc_now()
            await session.commit()
            logger.error("opportunity event {} failed: {}", event.id, exc)
            return True
        event.status = "DONE"
        event.processed_at = utc_now()
        await session.commit()
        return True
```

```python
async def rescore_lead(session: AsyncSession, business_id: UUID) -> None:
    """Fetch business + latest audit → detect opportunities → calc ROI → score → upsert."""
    business = await session.get(Business, business_id)
    if not business or not business.website_url:
        return

    # Latest completed audit
    audit = (
        await session.execute(
            select(WebsiteAudit)
            .where(WebsiteAudit.business_id == business_id, WebsiteAudit.status == "COMPLETED")
            .order_by(WebsiteAudit.started_at.desc())
            .limit(1)
        )
    ).scalars().first()

    # Instantiate engine components (config-driven)
    rule_engine = RuleEngine(settings.opportunity_rules_path)
    roi_calc = ROICalculator(settings.HOURLY_COST_BUCKETS, settings.IMPL_COST_TEMPLATES)
    scorer = LeadScorer(settings.SCORING_WEIGHTS["default"])

    # Detect opportunities
    auto_opps = rule_engine.evaluate_automation(business, audit) if audit else []
    ai_opps = rule_engine.evaluate_ai(business, audit) if audit else []
    all_opps = auto_opps + ai_opps

    # Add ROI to each
    for opp in all_opps:
        opp["roi_monthly_inr"] = roi_calc.calc_roi(opp, business, audit)
    all_opps.sort(key=lambda o: o["roi_monthly_inr"], reverse=True)

    # Score
    score, tier, breakdown = scorer.score_lead(business, audit, all_opps)

    # Upsert on Business (JSONB + scalar fields) — ONE commit by caller
    business.opportunities = all_opps
    business.opportunity_version = (business.opportunity_version or 0) + 1
    business.priority_score = score
    business.tier = tier
    business.score_breakdown = [b.__dict__ for b in breakdown]
    business.last_scored_at = datetime.now(UTC)
    business.total_roi = sum(o["roi_monthly_inr"] for o in all_opps)

    await AuditService.record(
        session, actor="system", action="lead_rescored",
        entity_type="business", entity_id=str(business_id),
        payload={"score": score, "tier": tier, "opportunity_count": len(all_opps)},
    )
```

## Shared Patterns

### Outbox Worker (Event-Driven Serial Polling)
**Source:** `backend/services/discovery/worker.py` (lines 24-69) + `backend/services/audits/worker.py` (lines 26-71)
**Apply to:** `backend/services/opportunities/worker.py`
**Pattern:** `while True: try: await _process_one(); except CancelledError: raise; except Exception: logger.exception(...); await asyncio.sleep(poll_interval)`. `_process_one`: select PENDING → mark IN_PROGRESS → commit → execute → DONE/FAILED + last_error → commit. Never crashes the loop.

### Runner/Engine (Batch Orchestration, One Transaction)
**Source:** `backend/services/audits/runner.py` (lines 199-325) + `backend/services/discovery/runner.py` (lines 139-219)
**Apply to:** `backend/services/opportunities/engine.py:rescore_lead`
**Pattern:** Session guard → load row → audit `*_started` → compute phases → upsert → audit `*_finished` → ONE `session.commit()`. Errors → `_mark_failed` with `last_error` + audit `*_failed`/`*_finished` → commit.

### Structured View Models + Jinja2 Autoescape
**Source:** `backend/api/discovery.py` (lines 54-68 status labels, 153-177 `_to_run_view`, 203-225 `_to_lead_view`, 336-357 TemplateResponse) + `backend/templates/discovery.html`
**Apply to:** `backend/api/discovery.py` (extended), `backend/templates/discovery.html` (extended)
**Pattern:** `_STATUS_LABELS` dicts for enums → `_to_*_view` functions return plain dicts with pre-formatted strings → templates render only view dicts, never ORM rows or raw JSONB. No `|safe` on scraped fields (CMP-06).

### Owner Auth + CSRF on Mutating POSTs
**Source:** `backend/auth/owner.py:10-19` (`Depends(get_current_owner)`) + `backend/api/discovery.py:73-80` (`_check_origin`) + `backend/main.py:109-116` (401 → `/auth/login`)
**Apply to:** `backend/api/discovery.py` (both `POST /run` and new `POST /rescore/{id}`)
**Pattern:** Every mutating route has `_owner: str = Depends(get_current_owner)` + `_check_origin(request)` at start. Flash + 302 redirect on success.

### Audit Trail with Correlation ID
**Source:** `backend/services/audit.py:16-37` (`AuditService.record`) + `backend/api/discovery.py:383-391` (trigger) + `backend/services/discovery/runner.py:155-163` (run_started)
